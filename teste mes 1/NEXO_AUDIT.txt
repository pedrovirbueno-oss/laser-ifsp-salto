  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),

  email VARCHAR(255) NOT NULL,
  phone VARCHAR(50) NOT NULL,

  created_at TIMESTAMPTZ DEFAULT NOW(),

import { NextResponse, NextRequest } from 'next/server';

export function middleware(req: NextRequest) {
  const isDemoMode = process.env.NEXT_PUBLIC_DEMO_MODE !== 'false';
  const pathname = req.nextUrl.pathname;

  // Static assets and API routes pass through to route-level handler security
  if (
    pathname.startsWith('/_next') ||
    pathname.startsWith('/static') ||
    pathname.startsWith('/api') ||
    pathname === '/' ||
    pathname === '/login'
  ) {
    return NextResponse.next();
  }

  // In Demo Mode, navigation is allowed (role switching is handled interactively)
  if (isDemoMode) {
    return NextResponse.next();
  }

  // In Production Mode, validate session token from cookies
  const authCookie = req.cookies.get('sb-access-token') || req.cookies.get('supabase-auth-token');

  if (!authCookie) {
    const loginUrl = new URL('/login', req.url);
    loginUrl.searchParams.set('redirect', pathname);
    return NextResponse.redirect(loginUrl);
  }

  // Role checks for restricted routes:
  // Note: Deep payload role validation also occurs inside server actions / API endpoints
  return NextResponse.next();
}

import { createServerClient } from '@supabase/ssr';
import { cookies } from 'next/headers';

export function createServerSupabaseClient() {
  const cookieStore = cookies();

  const supabaseUrl = process.env.NEXT_PUBLIC_SUPABASE_URL || 'https://placeholder.supabase.co';
  const supabaseAnonKey = process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY || 'placeholder-key';

  return createServerClient(supabaseUrl, supabaseAnonKey, {
    cookies: {
      get(name: string) {
        return cookieStore.get(name)?.value;
      },
      set(name: string, value: string, options: any) {
        try {
          cookieStore.set({ name, value, ...options });
        } catch (error) {
          // Handled in middleware/server component lifecycle
        }
      },
      remove(name: string, options: any) {
        try {
          cookieStore.set({ name, value: '', ...options });
        } catch (error) {
          // Handled in middleware/server component lifecycle
        }
      },
    },
  });
}

import { OeeMetrics, ProductionRecord, DowntimeRecord, Product, Target } from '@/types';

/**
 * Calculates OEE (Overall Equipment Effectiveness) for a machine or company dataset.
 * Formulas:
 *  - Availability = Run Time / Planned Operational Time
 *  - Performance  = Real Production Output / Theoretical Production Output (Target per hour * Run Time in hours)
 *  - Quality      = Good Units (Total - Scrap) / Total Units Produced
 *  - OEE          = Availability * Performance * Quality
 * 
 * Rules:
 *  - Clamped strictly between 0% and 100%.
 *  - Returns `sufficientData: false` when total planned time is 0 or no production records exist.
 */
export function calculateOee(
  productionRecords: ProductionRecord[],
  downtimeRecords: DowntimeRecord[],
  plannedTimeMinutes: number,
  targetProductsMap?: Record<string, Product>
): OeeMetrics {
  if (plannedTimeMinutes <= 0 || productionRecords.length === 0) {
    return {
      availability: 0,
      performance: 0,
      quality: 0,
      oee: 0,
      plannedTimeMinutes,
      downtimeMinutes: 0,
      runTimeMinutes: 0,
      totalCount: 0,
      scrapCount: 0,
      goodCount: 0,
      timeWindow: 'Período Selecionado',
      sufficientData: false,
    };
  }

  // Total Downtime in Minutes
  const totalDowntimeSeconds = downtimeRecords.reduce((acc, curr) => acc + (curr.duration_seconds || 0), 0);
  const downtimeMinutes = totalDowntimeSeconds / 60;

  // Run Time = Planned Time - Downtime (Cannot be negative)
  const runTimeMinutes = Math.max(0, plannedTimeMinutes - downtimeMinutes);

  // Availability = Run Time / Planned Time
  const rawAvailability = plannedTimeMinutes > 0 ? (runTimeMinutes / plannedTimeMinutes) : 0;

  // Production Totals
  let totalCount = 0;
  let scrapCount = 0;
  let theoreticalExpectedOutput = 0;

  for (const record of productionRecords) {
    const total = record.produced_quantity || 0;
    const scrap = record.rejected_quantity || 0;
    totalCount += total;
    scrapCount += scrap;

    // Calculate theoretical expected output if product target rates are available
    if (targetProductsMap && record.product_id && targetProductsMap[record.product_id]) {
      const product = targetProductsMap[record.product_id];
      const targetPerHour = product.target_quantity_per_hour || 60;
      
      // Calculate duration of this specific record in hours
      let durationHours = 0;
      if (record.start_time && record.end_time) {
        const start = new Date(record.start_time).getTime();
        const end = new Date(record.end_time).getTime();
        durationHours = Math.max(0, (end - start) / (1000 * 3600));
      } else {
        // Fallback: estimate runtime slice
        durationHours = (runTimeMinutes / 60) / Math.max(1, productionRecords.length);
      }
      theoreticalExpectedOutput += targetPerHour * durationHours;
    }
  }

  const goodCount = Math.max(0, totalCount - scrapCount);

  // Quality = Good Count / Total Count
  const rawQuality = totalCount > 0 ? (goodCount / totalCount) : 0;

  // Performance = Total Count / Theoretical Expected Output (or 100% if expected output is unknown)
  let rawPerformance = 1.0;
  if (theoreticalExpectedOutput > 0) {
    rawPerformance = totalCount / theoreticalExpectedOutput;
  } else if (runTimeMinutes > 0) {
    // Standard default assumption: 60 units/hr unless overridden
    const fallbackExpected = (runTimeMinutes / 60) * 60;
    rawPerformance = fallbackExpected > 0 ? (totalCount / fallbackExpected) : 1.0;
  }

  // Clamp Availability, Performance, Quality strictly between 0 and 1 (0% to 100%)
  const availabilityPct = Math.min(100, Math.max(0, rawAvailability * 100));
  const performancePct  = Math.min(100, Math.max(0, rawPerformance * 100));
  const qualityPct      = Math.min(100, Math.max(0, rawQuality * 100));

  // OEE = (Availability/100) * (Performance/100) * (Quality/100) * 100
  const oeePct = (availabilityPct / 100) * (performancePct / 100) * (qualityPct / 100) * 100;

  return {
    availability: Number(availabilityPct.toFixed(1)),
    performance: Number(performancePct.toFixed(1)),
    quality: Number(qualityPct.toFixed(1)),
    oee: Number(oeePct.toFixed(1)),
    plannedTimeMinutes: Math.round(plannedTimeMinutes),
    downtimeMinutes: Math.round(downtimeMinutes),
    runTimeMinutes: Math.round(runTimeMinutes),
    totalCount,
    scrapCount,
    goodCount,
    timeWindow: 'Período Selecionado',
    sufficientData: true,
  };
}

/**
 * Utility to compute total planned operational minutes for a given machine & period.
 * Takes into account scheduled shifts and days.
 */
export function calculatePlannedTimeMinutes(
  startDate: Date,
  endDate: Date,
  dailyShiftHours: number = 16 // Default: 2 shifts = 16 hrs/day
): number {
  const diffTime = Math.abs(endDate.getTime() - startDate.getTime());
  const diffDays = Math.max(1, Math.ceil(diffTime / (1000 * 60 * 60 * 24)));
  return diffDays * dailyShiftHours * 60;
}

import { MachineStatus, ProductionRecord, DowntimeRecord, DowntimeCategory } from '@/types';

export interface DynamicMachineState {
  status: MachineStatus;
  statusLabel: string;
  statusColor: 'green' | 'yellow' | 'blue' | 'red' | 'gray';
  statusBadge: string;
  activeDowntime?: DowntimeRecord;
  activeProduction?: ProductionRecord;
}

/**
 * Derives machine status dynamically from real-time events.
 * 
 * Priority Rules:
 *  1. Open Downtime Record (end_time is null/undefined):
 *     - category 'setup' -> 'setup' (🔵 Setup)
 *     - category 'maintenance' -> 'maintenance' (🔴 Manutenção)
 *     - all other downtime categories -> 'stopped' (🟡 Parada)
 *  2. Open Production Record (end_time is null/undefined) -> 'producing' (🟢 Produzindo)
 *  3. Neither -> 'inactive' (⚪ Sem Operação)
 */
export function deriveMachineStatus(
  activeProduction?: ProductionRecord | null,
  activeDowntime?: DowntimeRecord | null
): DynamicMachineState {
  if (activeDowntime && !activeDowntime.end_time) {
    const category = activeDowntime.category;
    if (category === 'setup') {
      return {
        status: 'setup',
        statusLabel: 'Em Setup',
        statusColor: 'blue',
        statusBadge: '🔵 Setup',
        activeDowntime,
      };
    }
    if (category === 'maintenance') {
      return {
        status: 'maintenance',
        statusLabel: 'Em Manutenção',
        statusColor: 'red',
        statusBadge: '🔴 Manutenção',
        activeDowntime,
      };
    }
    return {
      status: 'stopped',
      statusLabel: 'Parada',
      statusColor: 'yellow',
      statusBadge: '🟡 Parada',
      activeDowntime,
    };
  }

  if (activeProduction && !activeProduction.end_time) {
    return {
      status: 'producing',
      statusLabel: 'Produzindo',
      statusColor: 'green',
      statusBadge: '🟢 Produzindo',
      activeProduction,
    };
  }

  return {
    status: 'inactive',
    statusLabel: 'Sem Operação',
    statusColor: 'gray',
    statusBadge: '⚪ Sem Operação',
  };
}

import { AuditAction, AuditLog } from '@/types';

/**
 * Creates a structured audit log entry payload.
 * Sanitizes input to ensure passwords, secrets, or tokens are never logged.
 */
export function createAuditLogEntry(
  companyId: string,
  userId: string,
  userName: string,
  action: AuditAction,
  entityType: string,
  entityId: string,
  oldData?: Record<string, any> | null,
  newData?: Record<string, any> | null
): AuditLog {
  const sanitize = (obj?: Record<string, any> | null) => {
    if (!obj) return null;
    const sanitized = { ...obj };
    delete sanitized.password;
    delete sanitized.secret;
    delete sanitized.token;
    delete sanitized.key_hash;
    return sanitized;
  };

  return {
    id: typeof crypto !== 'undefined' && crypto.randomUUID ? crypto.randomUUID() : `audit-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
    company_id: companyId,
    user_id: userId,
    user_name: userName,
    action,
    entity_type: entityType,
    entity_id: entityId,
    old_data: sanitize(oldData),
    new_data: sanitize(newData),
    created_at: new Date().toISOString(),
  };

import { NextRequest, NextResponse } from 'next/server';
import { repositories } from '@/repositories';

export async function POST(req: NextRequest) {
  try {
    const rawApiKey = req.headers.get('x-api-key');
    if (!rawApiKey) {
      return NextResponse.json({ error: 'Chave de API (x-api-key) não fornecida' }, { status: 401 });
    }

    // Compute key hash (or query by hash)
    const keyHash = `hash_${rawApiKey}`;
    const validKey = await repositories.apiKeys.validateKey(keyHash);

    if (!validKey || !validKey.active) {
      return NextResponse.json({ error: 'Chave de API inválida, revogada ou inativa' }, { status: 401 });
    }

    const companyId = validKey.company_id; // Company derived STRICTLY from API key binding

    const body = await req.json();
    const { machine_code, timestamp, metrics } = body;

    if (!machine_code || typeof machine_code !== 'string') {
      return NextResponse.json({ error: 'Payload inválido: machine_code é obrigatório e deve ser uma string' }, { status: 400 });
    }

    if (!metrics || typeof metrics !== 'object') {
      return NextResponse.json({ error: 'Payload inválido: métricas de telemetria ausentes ou malformatadas' }, { status: 400 });
    }

    // Verify machine exists and belongs to the API key's company
    const machine = await repositories.machines.getByCode(companyId, machine_code);
    if (!machine) {
      return NextResponse.json({ error: `Máquina '${machine_code}' não encontrada para esta empresa` }, { status: 404 });
    }

    return NextResponse.json({
      success: true,
      message: 'Telemetria processada e autenticada com sucesso',
      company_id: companyId,
      machine_code,
      received_at: new Date().toISOString(),
    });
  } catch (err: any) {
    return NextResponse.json({ error: err.message || 'Erro interno no processamento de telemetria' }, { status: 500 });
  }
}

      status_label: state.statusLabel,

      active_downtime_reason: activeDown?.reason_id || null,
    });
  } catch (err: any) {
    return NextResponse.json({ error: err.message || 'Erro ao consultar status' }, { status: 500 });
  }

              <CheckCircle2 className="w-6 h-6 text-emerald-400" />

              <div className="p-4 bg-slate-950 border border-slate-800 rounded-lg space-y-2">
                <h3 className="font-bold text-blue-400">Empresa</h3>

                <h3 className="font-bold text-blue-400">Recursos Cadastrados</h3>
                <p className="text-xs text-slate-300">Máquinas: <strong className="text-white">{machines.length}</strong></p>
                <p className="text-xs text-slate-300">Produtos: <strong className="text-white">{products.length}</strong></p>
                <p className="text-xs text-slate-300">Turnos: <strong className="text-white">{shifts.length}</strong></p>
                <p className="text-xs text-slate-300">Operador: <strong className="text-white">{opName}</strong></p>
              </div>
            </div>
          </div>
        )}
      </div>

      {/* Footer Navigation Buttons */}
      <div className="max-w-4xl mx-auto w-full flex items-center justify-between border-t border-slate-800 pt-6 mt-6">
        {step > 1 ? (
          <button
            type="button"
            onClick={() => setStep(step - 1)}
            className="px-5 py-2.5 bg-slate-800 hover:bg-slate-700 text-slate-200 rounded-lg text-sm font-semibold flex items-center gap-2"
          >
            <ChevronLeft className="w-4 h-4" /> Voltar
          </button>
        ) : (
          <div />
        )}

        {step < 7 ? (
          <button
            type="button"
            onClick={() => setStep(step + 1)}
            className="px-6 py-2.5 bg-blue-600 hover:bg-blue-500 text-white rounded-lg text-sm font-semibold flex items-center gap-2 shadow-lg shadow-blue-500/20"
          >
            Avançar <ChevronRight className="w-4 h-4" />
          </button>
        ) : (
          <button
            type="button"
            disabled={isSubmitting}
            onClick={handleFinishSetup}
            className="px-8 py-3 bg-emerald-600 hover:bg-emerald-500 text-white rounded-lg text-sm font-bold flex items-center gap-2 shadow-lg shadow-emerald-500/20"
          >
            {isSubmitting ? 'Ativando...' : 'COMEÇAR A USAR O NEXO'}
          </button>
        )}
      </div>
    </div>
  );
}

'use client';

import React, { createContext, useContext, useEffect, useState } from 'react';
import { useRouter, usePathname } from 'next/navigation';
import { Profile, UserRole, Company } from '@/types';
import { repositories, IS_DEMO_MODE } from '@/repositories';
import { DEMO_COMPANY_ID } from '@/repositories/demo/seedData';

interface AuthContextType {
  user: Profile | null;
  company: Company | null;
  role: UserRole;
  isDemoMode: boolean;
  isLoading: boolean;
  switchDemoRole: (newRole: UserRole) => void;
  loginDemoUser: (email: string) => Promise<boolean>;
  logout: () => Promise<void>;
  checkFirstAccessSetup: () => Promise<boolean>;
}

const AuthContext = createContext<AuthContextType>({} as AuthContextType);

export const AuthProvider = ({ children }: { children: React.ReactNode }) => {
  const router = useRouter();
  const pathname = usePathname();
  const [user, setUser] = useState<Profile | null>(null);
  const [company, setCompany] = useState<Company | null>(null);
  const [role, setRole] = useState<UserRole>('admin');
  const [isLoading, setIsLoading] = useState<boolean>(true);

  useEffect(() => {
    async function initAuth() {
      try {
        if (IS_DEMO_MODE) {
          // Initialize Demo Admin profile
          const adminProf = await repositories.users.getByEmail('admin@nexoproduction.demo');
          if (adminProf) {
            setUser(adminProf);
            setRole(adminProf.role);
          }
          const comp = await repositories.company.getById(DEMO_COMPANY_ID);
          setCompany(comp);
        } else {
          // Production Supabase Auth check
          const { createClient } = await import('@/lib/supabase/client');
          const supabase = createClient();
          const { data: { session } } = await supabase.auth.getSession();
          
          if (session?.user) {
            const profile = await repositories.users.getById(session.user.id);
            if (profile) {
              setUser(profile);
              setRole(profile.role);
              const comp = await repositories.company.getById(profile.company_id);
              setCompany(comp);
            }
          }
        }
      } catch (err) {
        console.error('Auth initialization error:', err);
      } finally {
        setIsLoading(false);
      }
    }
    initAuth();
  }, []);

  const switchDemoRole = async (newRole: UserRole) => {
    if (!IS_DEMO_MODE) {
      console.warn('Role switching is disabled in Production Mode');
      return;
    }

    const emailMap: Record<UserRole, string> = {
      admin: 'admin@nexoproduction.demo',
      supervisor: 'supervisor@nexoproduction.demo',
      operator: 'operador@nexoproduction.demo',
    };

    const targetEmail = emailMap[newRole];
    const profile = await repositories.users.getByEmail(targetEmail);
    if (profile) {
      setUser(profile);
      setRole(profile.role);
      
      // Auto route to appropriate screen
      if (profile.role === 'operator' && pathname !== '/operator') {
        router.push('/operator');
      } else if (profile.role !== 'operator' && pathname === '/operator') {
        router.push('/dashboard');
      }
    }
  };

  const loginDemoUser = async (email: string): Promise<boolean> => {
    const profile = await repositories.users.getByEmail(email);
    if (profile) {
      setUser(profile);
      setRole(profile.role);
      const comp = await repositories.company.getById(profile.company_id);
      setCompany(comp);
      
      if (profile.role === 'operator') {
        router.push('/operator');
      } else {
        router.push('/dashboard');
      }
      return true;
    }
    return false;
  };

  const logout = async () => {
    if (!IS_DEMO_MODE) {
      const { createClient } = await import('@/lib/supabase/client');
      const supabase = createClient();
      await supabase.auth.signOut();
    }
    setUser(null);
    setCompany(null);
    router.push('/login');
  };

  const checkFirstAccessSetup = async (): Promise<boolean> => {
    if (!company) return false;
    const machines = await repositories.machines.getAll(company.id);
    const products = await repositories.products.getAll(company.id);
    const shifts = await repositories.company.getShifts(company.id);

    const isSetupComplete = machines.length >= 1 && products.length >= 1 && shifts.length >= 1;
    return isSetupComplete;
  };

  return (
    <AuthContext.Provider
      value={{
        user,
        company,
        role,
        isDemoMode: IS_DEMO_MODE,
        isLoading,
        switchDemoRole,
        loginDemoUser,
        logout,
        checkFirstAccessSetup,
      }}
    >
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => useContext(AuthContext);
